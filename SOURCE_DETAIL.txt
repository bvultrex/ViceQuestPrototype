func _ready() -> void:
    _configure_runtime_profile()
    _build_boot_ui()
    multiplayer.peer_connected.connect(_on_peer_connected)
    multiplayer.peer_disconnected.connect(_on_peer_disconnected)
    multiplayer.connected_to_server.connect(_on_connected_to_server)
    multiplayer.connection_failed.connect(_on_connection_failed)
    multiplayer.server_disconnected.connect(_on_server_disconnected)
    call_deferred("_run_boot_sequence")

func _exit_tree() -> void:
    if upnp_thread != null:
        var result_variant: Variant = upnp_thread.wait_to_finish()
        upnp_thread = null
        if result_variant is Dictionary and bool((result_variant as Dictionary).get("ok", false)):
            var result_gateway: Variant = (result_variant as Dictionary).get("gateway")
            if result_gateway is UPNP:
                (result_gateway as UPNP).delete_port_mapping(PORT, "UDP")
                return
    if upnp_mapping_owned and upnp_gateway != null:
        upnp_gateway.delete_port_mapping(PORT, "UDP")

func _run_boot_sequence() -> void:
    _set_boot_progress(0.04, "Selecting runtime profile")
    _configure_runtime_profile()
    combat_fx = COMBAT_FX_SCRIPT.new() as ViceQuestCombatFX
    combat_fx.name = "CombatFX"
    combat_fx.configure(runtime_low_power)
    add_child(combat_fx)
    audio_manager = AUDIO_MANAGER_SCRIPT.new() as ViceQuestAudioManager
    audio_manager.name = "GTA2Audio"
    audio_manager.configure(runtime_low_power)
    add_child(audio_manager)
    await get_tree().process_frame
    _set_boot_progress(0.14, "Reading Downtown surface and route banks")
    _load_downtown_surface_data()
    _load_population_route_banks()
    population_rng.seed = 0xC0FFEE42
    await get_tree().process_frame
    _set_boot_progress(0.28, "Building Downtown world chunks")
    _build_environment()
    _build_city()
    await get_tree().process_frame
    _set_boot_progress(0.48, "Placing workshops, missions and vehicles")
    _build_tuning_shops()
    _build_quest_markers()
    _build_vehicles()
    _build_police_pursuit_pool()
    await get_tree().process_frame
    _set_boot_progress(0.66, "Distributing GTA2 weapons and ammunition")
    _build_ammo_pickups()
    _build_weapon_pickups()
    await get_tree().process_frame
    _set_boot_progress(0.80, "Preparing streamed population and police")
    _build_cops()
    _build_civilians()
    await get_tree().process_frame
    _set_boot_progress(0.93, "Preparing camera and multiplayer menu")
    _build_camera()
    _build_ui()
    await get_tree().process_frame
    _set_boot_progress(1.0, "Downtown ready")
    boot_loading_panel.visible = false
    boot_splash_panel.visible = true

    boot_layer.visible = false
    boot_complete = true
    menu_panel.visible = true

    _start_room_browser()

func _add_fullscreen_ui_texture(parent: Control, texture_path: String) -> TextureRect:
    var image: TextureRect = TextureRect.new()
    image.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
    var selected_path: String = texture_path
    if runtime_low_power:
        var quest_path: String = "%s_quest.%s" % [texture_path.get_basename(), texture_path.get_extension()]
        if ResourceLoader.exists(quest_path):
            selected_path = quest_path
    image.texture = load(selected_path) as Texture2D
    image.expand_mode = TextureRect.EXPAND_IGNORE_SIZE
    # The supplied artwork is 4:3 while the prototype viewport is 16:9. Keep
    # the whole composition visible so the logo/loading caption are never
    # cropped; the parent ColorRect provides intentional side letterboxing.
    image.stretch_mode = TextureRect.STRETCH_KEEP_ASPECT_CENTERED
    image.texture_filter = CanvasItem.TEXTURE_FILTER_LINEAR
    image.mouse_filter = Control.MOUSE_FILTER_IGNORE
    parent.add_child(image)
    return image

func _set_boot_progress(value: float, message: String) -> void:
    if boot_progress != null:
        boot_progress.value = clampf(value, 0.0, 1.0) * 100.0
    if boot_status_label != null:
        boot_status_label.text = message

static var _smoke_texture_cache: Texture2D
static var _vehicle_texture_cache: Dictionary = {}

static func _vehicle_texture(texture_path: String) -> Texture2D:
    if _vehicle_texture_cache.has(texture_path):
        return _vehicle_texture_cache[texture_path] as Texture2D
    var texture: Texture2D = load(texture_path) as Texture2D
    _vehicle_texture_cache[texture_path] = texture
    return texture

func _ready() -> void:
    motion_mode = CharacterBody3D.MOTION_MODE_FLOATING
    collision_layer = 1
    collision_mask = 1
    if defer_runtime_resources:
        collision_layer = 0
        collision_mask = 0
    else:
        _ensure_runtime_resources()

func _ensure_runtime_resources() -> void:
    if _collision == null:
        _build_collision()
    if _sprite == null or _name_label == null:
        _build_visuals()
    set_variant(variant_id)

func _release_runtime_resources() -> void:
    if not ambient_pool_member or driver_id != 0:
        return
    if _smoke != null:
        _smoke.queue_free()
        _smoke = null
    if _name_label != null:
        _name_label.queue_free()
        _name_label = null
    if _sprite != null:
        _sprite.queue_free()
        _sprite = null
    if _collision != null:
        _collision.queue_free()
        _collision = null
    if _turret_pivot != null:
        _turret_pivot.queue_free()
        _turret_pivot = null
        _turret_sprite = null

func set_variant(new_variant_id: String) -> void:
    var was_full_health: bool = hp >= max_hp
    var clean: String = new_variant_id.to_lower()
    var texture_path: String = "res://assets/vehicles/%s.png" % clean
    if not ResourceLoader.exists(texture_path):
        clean = "jefferson"
        texture_path = "res://assets/vehicles/jefferson.png"
    variant_id = clean
    display_name = clean.replace("_", " ").to_upper()
    var physics: Dictionary = VEHICLE_DATA.definition(variant_id)
    max_forward_speed = float(physics["max_speed"])
    max_reverse_speed = float(physics["reverse_speed"])
    acceleration = float(physics["acceleration"])
    brake_deceleration = float(physics["braking"])
    max_hp = int(physics["max_hp"])
    collision_mass = float(physics.get("mass", 16.0))
    hp = max_hp if was_full_health else mini(hp, max_hp)
    _apply_damage_visuals()
    _refresh_tank_turret()
    _refresh_label()

func _configure_variant_size(wreck: bool) -> void:
    if _sprite != null:
        if wreck:
            _sprite.pixel_size = SPRITE_PIXEL_SIZE
        else:
            match variant_id:
                "tank":
                    _sprite.pixel_size = SPRITE_PIXEL_SIZE * 1.18
                "bus":
                    _sprite.pixel_size = SPRITE_PIXEL_SIZE * 1.30
                "box_truck", "benson", "hot_dog_van", "ice_cream_van", "van", "tow_truck", "truck_cab_sx", "sports_limousine", "pacifier", "dementia":
                    _sprite.pixel_size = SPRITE_PIXEL_SIZE * 1.12
                _:
                    _sprite.pixel_size = SPRITE_PIXEL_SIZE
    if _collision == null or not (_collision.shape is BoxShape3D):
        return
    var shape: BoxShape3D = _collision.shape as BoxShape3D
    match variant_id:
        "tank":
            shape.size = Vector3(2.38, 0.62 if not wreck else 0.22, 1.12)
        "bus":
            shape.size = Vector3(2.60, 0.22 if wreck else 0.55, 0.92)
        "box_truck", "benson", "hot_dog_van", "ice_cream_van", "van", "tow_truck", "truck_cab_sx", "sports_limousine", "pacifier", "dementia":
            shape.size = Vector3(2.05, 0.22 if wreck else 0.54, 0.84)
        _:
            shape.size = Vector3(1.72, 0.22 if wreck else 0.52, 0.78)

func _damage_texture_path() -> String:
    if is_destroyed:
        var wreck_index: int = int(WRECK_BY_VARIANT.get(variant_id, 3))
        return "res://assets/vehicles/wrecks/wreck_%d.png" % wreck_index
    return "res://assets/vehicles/%s.png" % variant_id

    vehicle_id = id
    spawn_position = start_position
    spawn_heading = start_heading
    position = start_position
    last_motion_origin = start_position
    last_impact_speed = 0.0
    heading = start_heading
    target_position = position
    target_heading = heading
    turret_heading = heading
    target_turret_heading = heading
    rotation.y = -heading

func _refresh_tank_turret() -> void:
    if variant_id != "tank":
        if _turret_pivot != null:
            _turret_pivot.visible = false
        return
    if _turret_pivot == null:
        _turret_pivot = Node3D.new()
        _turret_pivot.name = "TankTurret"
        # GTA2 model 148 is a separate OBJECT/CODE sprite. Keep the pivot at
        # the cupola centre so the original barrel rotates independently from
        # the vehicle body instead of orbiting around the texture centre.
        _turret_pivot.position.y = 0.11
        _turret_sprite = Sprite3D.new()
        _turret_sprite.name = "OriginalGTA2Turret"
        _turret_sprite.texture_filter = BaseMaterial3D.TEXTURE_FILTER_NEAREST
        _turret_sprite.billboard = BaseMaterial3D.BILLBOARD_DISABLED
        _turret_sprite.pixel_size = SPRITE_PIXEL_SIZE * 1.18
        _turret_sprite.rotation_degrees.x = -90.0
        # Exported texture is 92 px wide. Original cupola centre is around
        # x=36, while Sprite3D rotates about x=46, hence the +10 px correction.
        _turret_sprite.position.x = 10.0 * _turret_sprite.pixel_size
        _turret_pivot.add_child(_turret_sprite)
        add_child(_turret_pivot)
    if _turret_sprite != null:
        var turret_path: String = "res://assets/vehicles/tank_turret_damaged.png" if damage_stage >= 2 else "res://assets/vehicles/tank_turret.png"
        if ResourceLoader.exists(turret_path):
            _turret_sprite.texture = _vehicle_texture(turret_path)
        _turret_sprite.visible = not is_destroyed
    _turret_pivot.visible = stream_active and not is_destroyed
    _turret_pivot.rotation.y = heading - turret_heading

func configure_traffic(walk_route: Array[Vector3], owned: bool = true, cruise_speed: float = 6.2, loop_route: bool = true) -> void:
    ai_route = walk_route.duplicate()
    civilian_owned = owned
    ai_controlled = not ai_route.is_empty()
    stolen_once = false
    ai_route_index = 0
    if not ai_route.is_empty():
        var nearest_index: int = 0
        var nearest_distance: float = INF
        for i in range(ai_route.size()):
            var distance: float = Vector2(position.x, position.z).distance_to(Vector2(ai_route[i].x, ai_route[i].z))
            if distance < nearest_distance:
                nearest_distance = distance
                nearest_index = i
        ai_route_index = (nearest_index + 1) % ai_route.size()
    ai_cruise_speed = minf(cruise_speed, max_forward_speed * 0.88)
    ai_route_loop = loop_route
    _refresh_label()

func claim_by_player() -> bool:
    var theft: bool = civilian_owned and not stolen_once
    ai_controlled = false
    if theft:
        stolen_once = true
    _refresh_label()
    return theft

func _snap_to_map_surface() -> void:
    var main_node: Node = get_parent()
    if main_node != null and main_node.has_method("sample_surface_height"):
        var floor_y: float = float(main_node.call("sample_surface_height", position, position.y - VEHICLE_Y))
        position.y = floor_y + VEHICLE_Y

func server_ai_step(delta: float) -> void:
    if not multiplayer.is_server() or not stream_active or is_destroyed or not ai_controlled or driver_id != 0:
        return
    last_motion_origin = position
    last_impact_speed = 0.0
    if pursuit_active:
        _server_pursuit_step(delta)
        return
    if ai_route.is_empty():
        return

    var waypoint: Vector3 = ai_route[ai_route_index]
    var to_waypoint: Vector3 = waypoint - position
    to_waypoint.y = 0.0
    if to_waypoint.length() < 0.55:
        if not ai_route_loop and ai_route_index >= ai_route.size() - 1:
            current_speed = 0.0
            velocity = Vector3.ZERO
            if ambient_pool_member:
                # Route boundaries are not visual despawn points. The population manager
                # either stitches a next Green-Arrow segment or recycles us only when safe.
                var main_node: Node = get_parent()
                if main_node != null and main_node.has_method("_request_ambient_vehicle_recycle"):
                    main_node.call_deferred("_request_ambient_vehicle_recycle", vehicle_id)
            else:
                position = ai_route[0]
                target_position = position
                ai_route_index = 1 if ai_route.size() > 1 else 0
            return
        ai_route_index = (ai_route_index + 1) % ai_route.size()
        waypoint = ai_route[ai_route_index]
        to_waypoint = waypoint - position
        to_waypoint.y = 0.0

    if to_waypoint.length_squared() <= 0.01:
        current_speed = 0.0
        velocity = Vector3.ZERO
        return

    var previous_position: Vector3 = position
    # Traffic routes are axis-aligned lane-center paths. AI velocity is locked to
    # the current road axis instead of steering directly at a distant waypoint.
    # This prevents diagonal lane-cutting through intersections.
    var lane_direction: Vector3 = Vector3.ZERO
    if absf(to_waypoint.x) >= absf(to_waypoint.z):
        lane_direction = Vector3(1.0 if to_waypoint.x >= 0.0 else -1.0, 0.0, 0.0)
        position.z = move_toward(position.z, waypoint.z, 5.0 * delta)
    else:
        lane_direction = Vector3(0.0, 0.0, 1.0 if to_waypoint.z >= 0.0 else -1.0)
        position.x = move_toward(position.x, waypoint.x, 5.0 * delta)

    var desired_heading: float = atan2(lane_direction.z, lane_direction.x)
    heading = lerp_angle(heading, desired_heading, clampf(delta * 7.0, 0.0, 1.0))
    rotation.y = -heading
    panic_timer = maxf(0.0, panic_timer - delta)
    var desired_speed: float = ai_cruise_speed
    if panic_timer > 0.0:
        desired_speed = maxf(desired_speed, max_forward_speed * 0.94)
    current_speed = move_toward(current_speed, desired_speed, acceleration * (1.0 if panic_timer > 0.0 else 0.72) * delta)
    last_impact_speed = absf(current_speed)
    velocity = lane_direction * current_speed + _take_external_impulse_step(delta)
    move_and_slide()
    if _enforce_vehicle_surface(previous_position):
        _snap_to_map_surface()

    if get_slide_collision_count() > 0:
        current_speed *= 0.25
        _resolve_vehicle_collisions(impact_speed_for_ai())

    target_position = position
    target_heading = heading

func _build_collision() -> void:
    if _collision != null:
        return
    _collision = CollisionShape3D.new()
    var shape: BoxShape3D = BoxShape3D.new()
    shape.size = Vector3(1.72, 0.52, 0.78)
    _collision.shape = shape
    _collision.position.y = 0.28
    add_child(_collision)

    if _sprite == null:
        _sprite = Sprite3D.new()
        _sprite.texture = _vehicle_texture("res://assets/vehicles/jefferson.png")
        _sprite.texture_filter = BaseMaterial3D.TEXTURE_FILTER_NEAREST
        _sprite.billboard = BaseMaterial3D.BILLBOARD_DISABLED
        _sprite.pixel_size = SPRITE_PIXEL_SIZE
        _sprite.rotation_degrees.x = -90.0

    if _name_label == null:
        _name_label = Label3D.new()
        _name_label.billboard = BaseMaterial3D.BILLBOARD_ENABLED
        _name_label.position.y = 1.02
        _name_label.font_size = 18
        _name_label.outline_size = 7
        _name_label.pixel_size = 0.010
        _name_label.no_depth_test = true
        _name_label.visibility_range_end = 28.0
        add_child(_name_label)
    _refresh_tank_turret()
    _refresh_label()

func _ensure_smoke() -> void:
    if _smoke != null:
        return
    _smoke = GPUParticles3D.new()
    _smoke.amount = 10
    _smoke.lifetime = 1.35
    _smoke.randomness = 0.45
    _smoke.emitting = false
    _smoke.position = Vector3(0.0, 0.38, 0.0)

    var process: ParticleProcessMaterial = ParticleProcessMaterial.new()
    process.emission_shape = ParticleProcessMaterial.EMISSION_SHAPE_BOX
    process.emission_box_extents = Vector3(0.35, 0.04, 0.18)
    process.direction = Vector3(0.0, 1.0, 0.0)
    process.spread = 30.0
    process.gravity = Vector3(0.0, 0.12, 0.0)
    process.initial_velocity_min = 0.18
    process.initial_velocity_max = 0.42
    process.scale_min = 0.45
    process.scale_max = 0.95
    process.color = Color(0.22, 0.23, 0.24, 0.58)
    _smoke.process_material = process

    var quad: QuadMesh = QuadMesh.new()
    quad.size = Vector2(0.62, 0.62)
    var smoke_material: StandardMaterial3D = StandardMaterial3D.new()
    smoke_material.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA
    smoke_material.shading_mode = BaseMaterial3D.SHADING_MODE_UNSHADED
    smoke_material.billboard_mode = BaseMaterial3D.BILLBOARD_ENABLED
    smoke_material.albedo_texture = _make_smoke_texture()
    smoke_material.albedo_color = Color(0.32, 0.33, 0.34, 0.66)
    quad.material = smoke_material
    _smoke.draw_pass_1 = quad
    add_child(_smoke)

func _make_smoke_texture() -> Texture2D:
    if _smoke_texture_cache != null:
        return _smoke_texture_cache
    var image: Image = Image.create(32, 32, false, Image.FORMAT_RGBA8)
    for y in range(32):
        for x in range(32):
            var nx: float = (float(x) - 15.5) / 15.5
            var ny: float = (float(y) - 15.5) / 15.5
            var dist: float = sqrt(nx * nx + ny * ny)
            var wobble: float = sin(float(x) * 0.77 + float(y) * 1.13) * 0.045
            var alpha: float = clampf((1.0 - dist + wobble) * 0.78, 0.0, 0.62)
            image.set_pixel(x, y, Color(0.55, 0.55, 0.55, alpha))
    _smoke_texture_cache = ImageTexture.create_from_image(image)
    return _smoke_texture_cache

func set_ambient_pool_member(value: bool) -> void:
    ambient_pool_member = value
    _apply_damage_visuals()
    if ambient_pool_member and not stream_active:
        _release_runtime_resources()

func recycle_ambient(start_position: Vector3, start_heading: float, new_route: Array[Vector3], cruise_speed: float, variant: String) -> void:
    driver_id = 0
    current_speed = 0.0
    velocity = Vector3.ZERO
    hp = max_hp
    is_destroyed = false
    crushed_by_tank = false
    stolen_once = false
    panic_timer = 0.0
    external_impulse_velocity = Vector3.ZERO
    pursuit_active = false
    pursuit_role = ""
    reset_tuning()
    spawn_position = start_position
    spawn_heading = start_heading
    position = start_position
    last_motion_origin = start_position
    last_impact_speed = 0.0
    target_position = start_position
    heading = start_heading
    target_heading = start_heading
    rotation.y = -heading
    configure_traffic(new_route, true, cruise_speed, false)
    stream_active = true
    _ensure_runtime_resources()
    set_variant(variant)
    set_process(true)
    _apply_damage_visuals()
    _refresh_label()

func set_stream_active(active: bool) -> void:
    if driver_id != 0:
        active = true
    if stream_active == active:
        return
    stream_active = active
    if stream_active:
        _ensure_runtime_resources()
        position = target_position
    set_process(stream_active)
    _apply_damage_visuals()
    _refresh_label()
    if not stream_active:
        _release_runtime_resources()

func set_driver(new_driver_id: int) -> void:
    driver_id = new_driver_id
    if driver_id != 0:
        stream_active = true
        _ensure_runtime_resources()
        set_process(true)
        ai_controlled = false
    _refresh_label()

func mark_tank_crushed(value: bool = true) -> void:
    crushed_by_tank = value
    if crushed_by_tank:
        current_speed = 0.0
        velocity = Vector3.ZERO
        external_impulse_velocity = Vector3.ZERO
    _apply_damage_visuals()

func set_damage_state(new_hp: int, destroyed: bool) -> void:
    var previous_hp: int = hp
    var was_destroyed: bool = is_destroyed
    hp = clampi(new_hp, 0, max_hp)
    is_destroyed = destroyed
    if is_destroyed:
        driver_id = 0
        current_speed = 0.0
        velocity = Vector3.ZERO
    _apply_damage_visuals()
    if _sprite != null and (hp < previous_hp or (is_destroyed and not was_destroyed)):
        _play_damage_response(is_destroyed)
    _refresh_label()

func _apply_damage_visuals() -> void:
    # Streaming sleeps AI/physics. Visibility stays independent so a bad stream
    # decision can never turn Downtown into an empty city again.
    var damaged_threshold_hp: int = roundi(float(max_hp) * 0.65)
    var smoke_threshold_hp: int = roundi(float(max_hp) * 0.35)
    damage_stage = 3 if is_destroyed else (2 if hp <= smoke_threshold_hp else (1 if hp <= damaged_threshold_hp else 0))
    var physically_present: bool = stream_active and not crushed_by_tank
    if _sprite != null:
        _sprite.visible = stream_active or not ambient_pool_member
        var texture_path: String = _damage_texture_path()
        if ResourceLoader.exists(texture_path):
            _sprite.texture = _vehicle_texture(texture_path)
        var damage_tint: Color = Color.WHITE
        match damage_stage:
            1:
                damage_tint = Color(0.82, 0.78, 0.70, 1.0)
            2:
                damage_tint = Color(0.58, 0.52, 0.46, 1.0)
        _sprite.modulate = Color(
            paint_color.r * damage_tint.r,
            paint_color.g * damage_tint.g,
            paint_color.b * damage_tint.b,
            1.0
        )
        _sprite.position.y = 0.025 if is_destroyed else 0.05
    _configure_variant_size(is_destroyed)
    if _collision != null:
        _collision.set_deferred("disabled", not physically_present)
    collision_layer = 1 if physically_present else 0
    collision_mask = 1 if physically_present else 0
    if physically_present and hp > 0 and hp <= smoke_threshold_hp:
        _ensure_smoke()
    if _smoke != null:
        _smoke.emitting = physically_present and hp > 0 and hp <= smoke_threshold_hp
        if hp <= 35 and hp > 0:
            _smoke.amount = 16
        else:
            _smoke.amount = 10
    _refresh_tank_turret()

func _play_damage_response(wreck: bool) -> void:
    if _sprite == null:
        return
    if _damage_tween != null and _damage_tween.is_valid():
        _damage_tween.kill()
    _sprite.scale = Vector3(1.12, 0.88, 1.0) if wreck else Vector3(1.06, 0.94, 1.0)
    _damage_tween = create_tween()
    _damage_tween.tween_property(_sprite, "scale", Vector3.ONE, 0.18 if wreck else 0.10)

func _refresh_label() -> void:
    if _name_label == null:
        return
    if not stream_active or is_destroyed:
        _name_label.text = ""
        _name_label.visible = false
    elif driver_id == 0 and not ai_controlled:
        _name_label.text = "%s // %d HP\n[E] ENTER" % [display_name, hp]
        _name_label.visible = true
    elif ai_controlled:
        _name_label.text = ""
        _name_label.visible = false
    else:
        _name_label.text = ""
        _name_label.visible = false

func server_step(delta: float, input_dir: Vector2) -> void:
    if not multiplayer.is_server():
        return
    last_motion_origin = position
    last_impact_speed = 0.0
    if not stream_active and driver_id == 0:
        velocity = Vector3.ZERO
        return
    _crash_damage_cooldown = maxf(0.0, _crash_damage_cooldown - delta)
    if is_destroyed:
        current_speed = 0.0
        velocity = Vector3.ZERO
        return

    var throttle: float = -input_dir.y
    var steering: float = input_dir.x

    if driver_id == 0:
        current_speed = move_toward(current_speed, 0.0, COAST_DECELERATION * delta)
    elif throttle > 0.05:
        if current_speed < 0.0:
            current_speed = move_toward(current_speed, 0.0, brake_deceleration * delta)
        else:
            current_speed = move_toward(current_speed, max_forward_speed, acceleration * delta)
    elif throttle < -0.05:
        if current_speed > 0.0:
            current_speed = move_toward(current_speed, 0.0, brake_deceleration * delta)
        else:
            current_speed = move_toward(current_speed, -max_reverse_speed, acceleration * 0.72 * delta)
    else:
        current_speed = move_toward(current_speed, 0.0, COAST_DECELERATION * delta)

    if driver_id != 0 and absf(steering) > 0.05:
        if variant_id == "tank":
            # Differential tracks can pivot without forward motion.
            heading += steering * TANK_TURN_RATE * delta
        elif absf(current_speed) > 0.10:
            var speed_factor: float = clampf(absf(current_speed) / 6.0, 0.28, 1.0)
            var reverse_factor: float = 1.0 if current_speed >= 0.0 else -1.0
            heading += steering * STEER_RATE * speed_factor * reverse_factor * delta
        heading = wrapf(heading, -PI, PI)

    rotation.y = -heading
    var forward: Vector3 = get_forward_vector()
    var impact_speed: float = absf(current_speed)
    last_impact_speed = impact_speed
    var previous_position: Vector3 = position
    velocity = forward * current_speed + _take_external_impulse_step(delta)
    move_and_slide()
    if _enforce_vehicle_surface(previous_position):
        _snap_to_map_surface()

    if get_slide_collision_count() > 0:
        if impact_speed >= CRASH_DAMAGE_MIN_SPEED and _crash_damage_cooldown <= 0.0:
            _pending_crash_damage = clampi(int(round((impact_speed - 7.0) * 4.0)), 4, 25)
            _crash_damage_cooldown = CRASH_DAMAGE_COOLDOWN
        current_speed *= 0.30
        _resolve_vehicle_collisions(impact_speed)

    target_position = position
    target_heading = heading
    _refresh_tank_turret()

func server_turret_step(delta: float, input_axis: float) -> void:
    if not multiplayer.is_server() or variant_id != "tank" or driver_id == 0 or is_destroyed:
        return

func is_runover_contact(target_world_position: Vector3, target_radius: float = 0.34) -> bool:
    var half_length: float = 0.86
    var half_width: float = 0.39
    if _collision != null and _collision.shape is BoxShape3D:
        var box: BoxShape3D = _collision.shape as BoxShape3D
        half_length = box.size.x * 0.5
        half_width = box.size.z * 0.5

    var segment_start: Vector2 = Vector2(last_motion_origin.x, last_motion_origin.z)
    var segment_end: Vector2 = Vector2(position.x, position.z)
    var target: Vector2 = Vector2(target_world_position.x, target_world_position.z)
    var motion: Vector2 = segment_end - segment_start
    var direction: Vector2 = Vector2(get_forward_vector().x, get_forward_vector().z)
    if motion.length_squared() > 0.0001:
        direction = motion.normalized()
    segment_start -= direction * half_length
    segment_end += direction * half_length
    var segment: Vector2 = segment_end - segment_start
    var length_squared: float = maxf(segment.length_squared(), 0.0001)
    var progress: float = clampf((target - segment_start).dot(segment) / length_squared, 0.0, 1.0)
    var closest: Vector2 = segment_start + segment * progress
    return closest.distance_to(target) <= half_width + target_radius + 0.14

func _server_pursuit_step(delta: float) -> void:
    var to_target: Vector3 = pursuit_target - position
    to_target.y = 0.0
    if to_target.length_squared() <= 0.04:
        current_speed = move_toward(current_speed, 0.0, brake_deceleration * delta)
        velocity = _take_external_impulse_step(delta)
        move_and_slide()
        return
    var desired_direction: Vector3 = to_target.normalized()
    var desired_heading: float = atan2(desired_direction.z, desired_direction.x)
    var turn_speed: float = STEER_RATE * (1.15 if pursuit_role == "police" else 0.92)
    heading = lerp_angle(heading, desired_heading, clampf(turn_speed * delta, 0.0, 1.0))
    rotation.y = -heading
    var forward: Vector3 = get_forward_vector()
    var alignment: float = clampf(forward.dot(desired_direction), -0.4, 1.0)
    var chase_speed: float = max_forward_speed * (0.96 if pursuit_role == "police" else 0.90)
    if alignment < 0.25:
        chase_speed *= 0.42
    current_speed = move_toward(current_speed, chase_speed, acceleration * delta)
    var impact_speed: float = absf(current_speed)
    last_impact_speed = impact_speed
    var previous_position: Vector3 = position
    velocity = forward * current_speed + _take_external_impulse_step(delta)
    move_and_slide()
    if _enforce_vehicle_surface(previous_position):
        _snap_to_map_surface()
    if get_slide_collision_count() > 0:
        if impact_speed >= CRASH_DAMAGE_MIN_SPEED and _crash_damage_cooldown <= 0.0:
            _pending_crash_damage = clampi(roundi((impact_speed - 6.5) * 3.2), 4, 22)
            _crash_damage_cooldown = CRASH_DAMAGE_COOLDOWN
        current_speed *= -0.18
        _resolve_vehicle_collisions(impact_speed)
    target_position = position
    target_heading = heading

func _enforce_vehicle_surface(previous_position: Vector3) -> bool:
    var main_node: Node = get_parent()
    if main_node == null or not main_node.has_method("is_vehicle_surface_allowed"):
        return true
    if bool(main_node.call("is_vehicle_surface_allowed", previous_position, position)):
        return true
    position = previous_position
    velocity = Vector3.ZERO
    external_impulse_velocity = Vector3.ZERO
    current_speed *= -0.16
    target_position = position
    return false

func _resolve_vehicle_collisions(impact_speed: float) -> void:
    for collision_index in range(get_slide_collision_count()):
        var collision: KinematicCollision3D = get_slide_collision(collision_index)
        var collider: Object = collision.get_collider()
        if not collider is ViceQuestVehicle:
            continue
        var other: ViceQuestVehicle = collider as ViceQuestVehicle
        if other == self or other.is_destroyed:
            continue
        if variant_id == "tank" and other.variant_id != "tank" and multiplayer.is_server():
            var main_node: Node = get_parent()
            if main_node != null and main_node.has_method("_server_tank_crush_vehicle"):
                main_node.call("_server_tank_crush_vehicle", vehicle_id, other.vehicle_id, driver_id)
            # The victim becomes non-collidable immediately. Preserve most tank
            # momentum and advance through the old contact plane so the tank
            # behaves as a heavy tracked vehicle instead of bouncing backward.
            var travel_sign: float = 1.0 if current_speed >= 0.0 else -1.0
            current_speed = travel_sign * maxf(absf(current_speed), impact_speed * 0.82)
            position += get_forward_vector() * travel_sign * clampf(impact_speed * 0.045, 0.38, 0.92)
            _snap_to_map_surface()
            target_position = position
            continue
        var normal: Vector3 = collision.get_normal()
        normal.y = 0.0
        if normal.length_squared() <= 0.001:
            normal = (position - other.position).normalized()
        else:
            normal = normal.normalized()
        var relative_speed: float = maxf(impact_speed, (velocity - other.velocity).length())
        if relative_speed < 1.2:
            continue
        var self_mass: float = clampf(collision_mass / 16.0, 0.35, 2.2)
        var other_mass: float = clampf(other.collision_mass / 16.0, 0.35, 2.2)
        var impulse: float = clampf(relative_speed * 0.48, 1.4, 9.5)
        apply_external_impulse(normal * impulse * other_mass / self_mass)
        other.apply_external_impulse(-normal * impulse * self_mass / other_mass)
        current_speed *= -0.22
        other.current_speed *= 0.55
        if _crash_damage_cooldown <= 0.0 and relative_speed >= CRASH_DAMAGE_MIN_SPEED:
            _pending_crash_damage = maxi(_pending_crash_damage, clampi(roundi(relative_speed * 1.6), 5, 24))
            _crash_damage_cooldown = CRASH_DAMAGE_COOLDOWN
        if other._crash_damage_cooldown <= 0.0 and relative_speed >= CRASH_DAMAGE_MIN_SPEED:
            other._pending_crash_damage = maxi(other._pending_crash_damage, clampi(roundi(relative_speed * 1.35), 4, 20))
            other._crash_damage_cooldown = CRASH_DAMAGE_COOLDOWN

func apply_external_impulse(impulse: Vector3) -> void:
    impulse.y = 0.0
    external_impulse_velocity = (external_impulse_velocity + impulse).limit_length(MAX_EXTERNAL_IMPULSE_SPEED)

    target_heading = heading

# GTA2-inspired lightweight sound layer. The original game continuously mixes

func configure(quest_low_power: bool) -> void:
    low_power = quest_low_power
    engine_player = AudioStreamPlayer3D.new()
    engine_player.name = "LocalVehicleEngine"
    engine_player.max_distance = 30.0
    engine_player.unit_size = 5.5
    engine_player.attenuation_filter_cutoff_hz = 11500.0
    engine_player.volume_db = -11.0

    wasted_player = AudioStreamPlayer.new()
    wasted_player.name = "WastedVoice"
    wasted_player.volume_db = -3.0
    if ResourceLoader.exists(WASTED_PATH):
        wasted_player.stream = load(WASTED_PATH)
    add_child(wasted_player)

    # Recycle short spatial voices instead of allocating a new node per shot.
    # This matters for automatic weapons and keeps the Quest sound layer bounded.
    var oneshot_count: int = 12 if low_power else 24
    for index in range(oneshot_count):
        var voice: AudioStreamPlayer3D = AudioStreamPlayer3D.new()
        voice.name = "OneShot_%02d" % index
        voice.unit_size = 5.0
        add_child(voice)
        _oneshot_pool.append(voice)

func _stream(sample_key: String) -> AudioStream:
    if _cache.has(sample_key):
        return _cache[sample_key] as AudioStream
    var path: String = SFX_ROOT + sample_key + ".wav"
    if not ResourceLoader.exists(path):
        return null
    var stream: AudioStream = load(path) as AudioStream
    _cache[sample_key] = stream
    return stream

func _looped_stream(sample_key: String) -> AudioStream:
    var base: AudioStream = _stream(sample_key)
    if base == null:
        return null
    if base is AudioStreamWAV:
        var looped: AudioStreamWAV = (base as AudioStreamWAV).duplicate() as AudioStreamWAV
        looped.loop_mode = AudioStreamWAV.LOOP_FORWARD
        return looped
    return base

func play_wasted() -> void:
    if wasted_player == null or wasted_player.stream == null:
        return
    wasted_player.stop()
    wasted_player.play()

func play_weapon(weapon_id: int, world_position: Vector3, tank_cannon: bool = false) -> void:

    if weapon_id == 11:
        pitch = 1.04
    _play_spatial(key, world_position, -5.0, pitch, 31.0)

    # first pass layers two authentic heavy/weapon samples rather than claiming

        key = "door_heavy_open" if opening else "door_heavy_close"
    else:
        key = "door_light_open" if opening else "door_light_close"

func update_local_vehicle(vehicle: ViceQuestVehicle) -> void:
    if engine_player == null:
        return
    if vehicle == null or vehicle.is_destroyed or vehicle.driver_id == 0:
        stop_engine()
        return
    var key: String = String(ENGINE_BY_VARIANT.get(vehicle.variant_id, "engine_standard"))
    if key != _engine_key or engine_player.stream == null:
        _engine_key = key
        engine_player.stream = _looped_stream(key)
        if engine_player.stream != null:
            engine_player.play()
    if engine_player.stream == null:
        return
    engine_player.global_position = vehicle.global_position
    var speed_ratio: float = clampf(absf(vehicle.current_speed) / maxf(vehicle.max_forward_speed, 0.1), 0.0, 1.0)
    engine_player.pitch_scale = 0.76 + speed_ratio * (0.48 if vehicle.variant_id == "tank" else 0.64)
    engine_player.volume_db = lerpf(-13.0, -5.0, speed_ratio)
    if not engine_player.playing:
        engine_player.play()

func stop_engine() -> void:
    _engine_key = ""
    if engine_player != null and engine_player.playing:
        engine_player.stop()

func _play_spatial(sample_key: String, world_position: Vector3, volume_db: float, pitch: float, max_distance: float) -> void:
    var stream: AudioStream = _stream(sample_key)
    if stream == null or _oneshot_pool.is_empty():
        return
    var player: AudioStreamPlayer3D = _oneshot_pool[_oneshot_cursor]
    _oneshot_cursor = (_oneshot_cursor + 1) % _oneshot_pool.size()
    player.stop()
    player.stream = stream
    player.global_position = world_position
    player.volume_db = volume_db
    player.pitch_scale = pitch
    player.max_distance = max_distance
    player.play()

            _visual.play_state("death", was_alive)
    _refresh_visuals()

func _refresh_visuals() -> void:
    # Keep the body visible after death so the death sequence can finish.
    visible = true
    var physically_present: bool = is_alive and not is_incapacitated()
    collision_layer = 1 if physically_present else 0
    collision_mask = 1 if physically_present else 0
    if _collision != null:
        _collision.set_deferred("disabled", not physically_present)
    if _label != null:
        _label.visible = is_alive
    _update_animation()

func reset_to_spawn() -> void:
    hp = 90
    is_alive = true
    position = spawn_position
    target_position = spawn_position
    velocity = Vector3.ZERO
    knockback_velocity = Vector3.ZERO
    unconscious_timer = 0.0
    recovery_timer = 0.0
    recovery_flee_timer = 0.0
    burning_timer = 0.0
    _remote_knockdown_state = 0
    _moving = false
    _armed = false
    _refresh_visuals()
    if _visual != null:
        _visual.play_state("idle", true)

func apply_remote_state(server_position: Vector3, server_facing: Vector3, alive: bool, new_hp: int, incapacitated_state: int = 0) -> void:
    target_position = server_position
    if server_facing.length_squared() > 0.0:
        facing = server_facing
    var was_alive: bool = is_alive
    hp = new_hp
    is_alive = alive
    _remote_knockdown_state = incapacitated_state
    if not is_alive and _visual != null:
        _visual.play_state("death", was_alive)
    _refresh_visuals()
    if _visual != null and is_alive:
        if _remote_knockdown_state == 1:
            _visual.show_frame("death", 16)
        elif _remote_knockdown_state == 2:
            _visual.show_frame("death", 6)

func _process(delta: float) -> void:
    if multiplayer.is_server():
        return
    var planar_distance: float = Vector2(position.x, position.z).distance_to(Vector2(target_position.x, target_position.z))
    _moving = planar_distance > 0.04
    var alpha: float = 1.0 - exp(-NETWORK_INTERPOLATION_SPEED * delta)
    position = position.lerp(target_position, alpha)
    if facing.length_squared() > 0.0:
        rotation.y = -atan2(facing.x, -facing.z)
    _update_animation()
